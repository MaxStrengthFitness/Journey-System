```powershell
cd C:\Users\austi\Projects\Journey-System-Beta-master\.claude\worktrees\colour-build
git revert --no-edit 916afbf0
npx tsc --noEmit
$env:TZ = "America/New_York"; npx vitest run --dir src --testTimeout=30000
```
